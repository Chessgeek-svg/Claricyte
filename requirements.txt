# Runtime dependencies for the demo and the training pipeline.
#
# torch and torchvision are pinned without a build suffix so that a plain
# `pip install -r requirements.txt` resolves on any platform (CI gets the CPU
# build). For GPU training, install them from the CUDA index FIRST, then run
# this file, which will leave the CUDA builds in place:
#   pip install torch==2.5.1 torchvision==0.20.1 \
#       --index-url https://download.pytorch.org/whl/cu121
#
# The CPU index below is for hosting. On Linux, PyPI's default torch wheel bundles
# the CUDA runtime and installs to several GB, which a small hosted container will
# not tolerate; the CPU build is a fraction of that and the demo never uses a GPU.
# A local version (2.5.1+cpu) sorts after the plain 2.5.1 under PEP 440, so pip
# prefers it from here. An existing 2.5.1+cu121 install already satisfies ==2.5.1,
# so this does not disturb a GPU machine.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.5.1
torchvision==0.20.1

timm==1.0.28

pandas==3.0.3
pillow==12.3.0
streamlit==1.60.0

# Retrieval. Needed at runtime because the demo answers questions in-app, but

# ~1GB, and the quiz path never touches them. See claricyte/rag/store.py.
chromadb==1.5.9
sentence-transformers==6.0.0

# Generation. Called only from the question box, never on the quiz path.
